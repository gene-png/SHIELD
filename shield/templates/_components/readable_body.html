{# Render an artifact body_text as readable content when we can recognize its shape, falling back to pretty-printed JSON or a plain 
.

   Inputs:
     body_text   raw string from the artifact (usually JSON)
     truncate    int, optional. Cap the body shown to N chars (default: full).

   Recognized shapes (best-effort):
     - {executive_summary: {...}, findings: [...]}   (P3 ATT&CK coverage)
     - {answer, citations}                           (P1 chat)
     - {gaps: [...], redundancies: [...]}            (P1 overlap)
     - [array of objects with `name`]                (P1 extraction)
     - {controls: [...]}                             (P2 posture/roadmap)
     - other JSON                                    pretty-printed JSON
     - non-JSON                                      plain 

   For unrecognized JSON we still show the structured fallback (the
   pretty-printed tree in 
), not just an opaque 
 dump.
#}
{%- set raw = body_text or "" -%}
{%- if truncate is not defined or not truncate -%}
  {%- set _show = raw -%}
{%- else -%}
  {%- set _show = raw[:truncate] -%}
{%- endif -%}

{%- set parsed = none -%}
{%- if _show.strip().startswith("{") or _show.strip().startswith("[") -%}
  {# We can't try/except in Jinja, so route through a tiny helper filter.
     `from_json` is registered in shield/__init__.py — it returns None
     on parse failure rather than raising. #}
  {%- set parsed = _show | from_json -%}
{%- endif -%}

{%- if parsed is mapping -%}
  {# ---- Shape: P3 ATT&CK coverage ---- #}
  {%- if parsed.get("executive_summary") or parsed.get("findings") -%}
    {%- set summary = parsed.get("executive_summary") or {} -%}
    {%- if summary.get("headline") -%}
      
{{ summary["headline"] }}

    {%- endif -%}
    {%- if summary.get("covered") is not none -%}
      
    
        
  • {{ summary.get("covered", 0) }} covered
    • 
        
  • {{ summary.get("partial", 0) }} partial
    • 
        
  • {{ summary.get("uncovered", 0) }} not covered
    • 
        
  • {{ summary.get("total_techniques", "") }} techniques evaluated
    • 
      

    {%- endif -%}
    {%- set findings = parsed.get("findings", []) -%}
    {%- if findings -%}
      
{{ findings | length }} findings — details below.

    {%- endif -%}

  {# ---- Shape: P1 chat ---- #}
  {%- elif parsed.get("answer") -%}
    
{{ parsed["answer"] }}

    {%- set cites = parsed.get("citations") or parsed.get("evidence") or [] -%}
    {%- if cites -%}
      
Cited:
        {%- for c in cites %}{{ c }}{%- if not loop.last %}; {% endif -%}{% endfor -%}
      

    {%- endif -%}

  {# ---- Shape: P1 overlap ---- #}
  {%- elif parsed.get("redundancies") or parsed.get("gaps") -%}
    {%- if parsed.get("redundancies") -%}
      
Redundancies

      
    
        {%- for r in parsed["redundancies"] -%}
          
  • 
            {%- if r is mapping -%}
              {{ r.get("category", "(group)") }}
              {%- if r.get("items") %}: {{ r.get("items") | join(", ") }}{% endif -%}
              {%- if r.get("note") %} — {{ r["note"] }}{% endif -%}
            {%- else -%}{{ r }}{%- endif -%}
          
    • 
        {%- endfor -%}
      

    {%- endif -%}
    {%- if parsed.get("gaps") -%}
      
Gaps

      
    
        {%- for g in parsed["gaps"] -%}
          
  • {% if g is mapping %}{{ g.get("description", g) }}{% else %}{{ g }}{% endif %}
    • 
        {%- endfor -%}
      

    {%- endif -%}

  {# ---- Shape: P2 posture / roadmap (controls list) ---- #}
  {%- elif parsed.get("controls") -%}
    
{{ parsed["controls"] | length }} controls assessed.

    {%- if parsed.get("summary") -%}
      
{{ parsed["summary"] }}

    {%- endif -%}

  {%- else -%}
    {# Recognized JSON shape we don't have a custom renderer for — let
       the pretty-print fallback below carry the load. #}
  {%- endif -%}

{%- elif parsed is sequence and parsed is not string -%}
  {# ---- Shape: P1 extraction (array of capability items) ---- #}
  
{{ parsed | length }} items extracted.

  {%- if parsed and parsed[0] is mapping and parsed[0].get("name") -%}
    
    
      {%- for item in parsed[:8] -%}
        
  • {{ item.get("name", "(unnamed)") }}
          {%- if item.get("vendor") %} · {{ item["vendor"] }}{% endif -%}
          {%- if item.get("category") %} · {{ item["category"] }}{% endif -%}
        
    • 
      {%- endfor -%}
      {%- if parsed | length > 8 -%}
        
  • … and {{ parsed | length - 8 }} more (see full content below).

      {%- endif -%}
    

  {%- endif -%}
{%- endif -%}
